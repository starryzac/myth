class VerificationError(ValueError):
    """A required actual proof is missing or inconsistent."""


def require(condition: bool, message: str) -> None:
    if not condition:
        raise VerificationError(message)


def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def encode(value: Any) -> str:
    # Exact full-table algorithm copied from verify_mvp304_audit_target.py.
    return json.dumps(value, ensure_ascii=False, sort_keys=True, default=str, separators=(",", ":"))


def read_file(path: Path) -> bytes:
    require(path.is_file() and not path.is_symlink(), f"Missing or symlinked file: {path}")
    require(path.stat().st_size <= 128 * 1024 * 1024, f"Artifact is too large: {path}")
    return path.read_bytes()


def checked_pins() -> dict[str, bytes]:
    result = {name: read_file(ROOT / name) for name in PINS}
    for name, value in result.items():
        require(sha(value) == PINS[name], f"Pinned baseline/tool changed: {name}")
    return result


def load_validator() -> ModuleType:
    # This module has a safe main guard and only stdlib top-level imports. Its
    # validate_manifest/verify(..., dry_run=True) read evidence and git inventory;
    # neither starts checks nor connects to any database. Check its pin first.
    require(sha(read_file(ROOT / VALIDATOR_PATH)) == PINS[VALIDATOR_PATH], "Validator changed")
    spec = importlib.util.spec_from_file_location("mvp304_completed_check", ROOT / VALIDATOR_PATH)
    require(spec is not None and spec.loader is not None, "Cannot load completed-check validator")
    assert spec is not None and spec.loader is not None
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def check_full_snapshot(snapshot: dict[str, Any], validator: ModuleType) -> None:
    validator.instant(snapshot.get("read_at"), "baseline read_at")
    require(snapshot.get("read_only") is True, "Snapshot did not observe READ ONLY")
    require(snapshot.get("isolation") == "repeatable read", "Snapshot did not observe RR")
    require(snapshot.get("database_revision") == REVISION, "Snapshot revision differs")
    data, columns = snapshot.get("data"), snapshot.get("columns")
    require(type(data) is dict and len(data) == 23, "Expected all actual 23 tables")
    require(type(columns) is dict and data.keys() == columns.keys(), "Column inventory differs")
    require(
        snapshot.get("table_counts") == {name: len(rows) for name, rows in data.items()},
        "Snapshot counts are not the actual row counts",
    )
    for name, rows in data.items():
        require(type(rows) is list, f"Invalid row list: {name}")
        require(type(columns[name]) is list and bool(columns[name]), f"Missing columns: {name}")
        require(len(set(columns[name])) == len(columns[name]), f"Duplicate columns: {name}")
        require(rows == sorted(rows, key=encode), f"Noncanonical row order: {name}")
        require(
            all(type(row) is dict and row.keys() == set(columns[name]) for row in rows),
            f"Snapshot row omits actual columns: {name}",
        )
    require(
        snapshot.get("table_sha256")
        == {name: sha(encode(rows).encode()) for name, rows in data.items()},
        "Snapshot per-table hashes are inconsistent",
    )
    require(snapshot.get("data_sha256") == sha(encode(data).encode()), "Snapshot hash differs")
    require(snapshot["data_sha256"] == FORMAL_DATA_HASH, "Unexpected formal baseline data")


def compare_full(actual: dict[str, Any], expected: dict[str, Any], label: str) -> None:
    for key in (
        "database_revision",
        "columns",
        "data",
        "table_counts",
        "table_sha256",
        "data_sha256",
    ):
        require(actual[key] == expected[key], f"Formal data differs from {label}: {key}")


def load_baselines(pinned: dict[str, bytes], validator: ModuleType) -> dict[str, Any]:
    before = validator.parse_json(pinned[BEFORE_PATH])
    after = validator.parse_json(pinned[AFTER_PATH])
    archive = validator.parse_json(pinned[ARCHIVE_PATH])
    migration = validator.parse_json(pinned[MIGRATION_PATH])
    check_full_snapshot(before, validator)
    check_full_snapshot(after, validator)
    compare_full(after, before, "negative before/after")
    formal = archive.get("formal_expected_negative")
    require(type(formal) is dict, "Missing archived formal-negative proof")
    require(formal.get("run_id") == FORMAL_RUN_ID, "Unexpected baseline command run")
    require(formal.get("database_name") == "bounded_funds", "Wrong formal baseline name")
    require(formal.get("cli_status") == "NOT_VERIFIED", "Formal legacy status was lost")
    require(formal.get("process_exit_code") == 1, "Formal legacy command did not exit nonzero")
    require(formal.get("command_exit_code") == 1, "Formal legacy manifest command differs")
    require(formal.get("manifest_successful") is False, "Formal negative became successful")
    require(formal.get("registered_epoch_count") == 0, "Formal baseline has audit history")
    require(formal.get("formal_seed_reset") is False, "Baseline includes formal reset")
    require(formal.get("formal_history_created") is False, "Baseline created formal history")
    for key in ("table_counts", "table_sha256"):
        require(formal.get(key) == before[key], f"Archive differs from actual snapshots: {key}")
    require(
        formal.get("before_data_sha256") == formal.get("after_data_sha256") == FORMAL_DATA_HASH,
        "Archived full-data hashes differ",
    )
    # Validate the archived command artifacts without rerunning that command.
    for name, expected in formal["archived_files_sha256"].items():
        require(Path(name).name == name, "Unsafe archive artifact path")
        require(
            sha(read_file(ROOT / "docs/progress/evidence" / name)) == expected,
            f"Archived negative artifact changed: {name}",
        )
    validator.instant(migration.get("read_at"), "migration read_at")
    require(migration.get("read_only") is True, "Migration observation was not RO")
    require(migration.get("isolation") == "repeatable read", "Migration observation was not RR")
    require(migration.get("database_revision") == REVISION, "Migration revision differs")
    require(
        len(migration["original_columns"]) == len(migration["data"]) == 20,
        "Migration lacks all original 20 tables",
    )
    require(
        migration["original_columns"].keys() == migration["data"].keys(),
        "Original migration table sets differ",
    )
    require(
        migration["data_sha256"] == sha(encode(migration["data"]).encode()) == ORIGINAL_DATA_HASH,
        "Migration original-column data hash differs",
    )
    require(
        migration["table_counts"] == {n: len(r) for n, r in migration["data"].items()},
        "Original migration row counts differ",
    )
    return {"before": before, "after": after, "formal": formal, "migration": migration}


def snapshot(engine: Any, baselines: dict[str, Any]) -> tuple[dict[str, Any], dict[str, Any]]:
    # SQLAlchemy is imported only after every targeted/source/baseline gate.
    from sqlalchemy import inspect, text

    with engine.connect().execution_options(isolation_level="REPEATABLE READ") as connection:
        with connection.begin():
            connection.execute(text("SET TRANSACTION READ ONLY"))
            isolation = connection.scalar(text("SHOW transaction_isolation"))
            read_only = connection.scalar(text("SHOW transaction_read_only")) == "on"
            database = connection.scalar(text("SELECT current_database()"))
            require(
                database == "bounded_funds", "Connected database is not original formal database"
            )
            require(read_only and isolation == "repeatable read", "Actual transaction is not RR/RO")
            inspector = inspect(connection)
            names = sorted(inspector.get_table_names())
            require(
                names == sorted(baselines["before"]["columns"]), "Actual table inventory changed"
            )
            quote = connection.dialect.identifier_preparer.quote
            data, columns = {}, {}
            for name in names:
                columns[name] = [column["name"] for column in inspector.get_columns(name)]
                rows = list(connection.scalars(text(f"SELECT to_jsonb(t) FROM {quote(name)} AS t")))
                data[name] = sorted(rows, key=encode)
            result = {
                "read_at": datetime.now(UTC).isoformat(),
                "database_name": database,
                "isolation": isolation,
                "read_only": read_only,
                "database_revision": connection.scalar(
                    text("SELECT version_num FROM alembic_version")
                ),
                "columns": columns,
                "data": data,
                "table_counts": {name: len(rows) for name, rows in data.items()},
                "table_sha256": {name: sha(encode(rows).encode()) for name, rows in data.items()},
                "data_sha256": sha(encode(data).encode()),
            }
            original_columns = baselines["migration"]["original_columns"]
            original = {}
            for name, fields in original_columns.items():
                require(
                    set(fields).issubset(columns[name]), f"Original columns disappeared: {name}"
                )
                statement = (
                    "SELECT "
                    + ",".join(quote(field) for field in fields)
                    + " FROM "
                    + quote(name)
                    + " ORDER BY id"
                )
                original[name] = [
                    dict(row) for row in connection.exec_driver_sql(statement).mappings()
                ]
            # Exact typed-column conversion copied from capture_mvp304_db.py.
            original_data = json.loads(json.dumps(original, default=str, sort_keys=True))
            original_result = {
                "original_columns": original_columns,
                "data": original_data,
                "table_counts": {name: len(rows) for name, rows in original_data.items()},
                "data_sha256": sha(encode(original_data).encode("utf-8")),
            }
            return result, original_result


def exclusive_write(path: Path, value: dict[str, Any]) -> str:
    require(not path.is_symlink() and not path.exists(), f"Refusing to replace evidence: {path}")
    data = (json.dumps(value, ensure_ascii=False, indent=2) + "\n").encode("utf-8")
    with path.open("xb") as output:
        output.write(data)
    require(read_file(path) == data, f"New evidence write differs: {path}")
    return sha(data)


def bound_file(relative: str) -> Path:
    require(type(relative) is str and bool(relative), "Missing evidence file path")
    path = (ROOT / relative).resolve()
    require(path.is_relative_to(ROOT), "Evidence path leaves the current repository")
    require(not (ROOT / relative).is_symlink(), "Evidence file is symlinked")
    return path


def targeted_gate(

    require(red.get("full_check_verified") is False, "Second failed full check became GREEN")
    require(red.get("command_exit_codes") == [0] * 8 + [1], "Failed command exits differ")
    require(
        red.get("actual_result")
        == {"collected": 1300, "passed": 464, "failed": 1, "warnings": 1, "seconds": 9263.84},
        "Actual failed full-check result differs",
    )
    archived = {}
    for item in red["archived"]:
        name = item["file"]
        require(
            Path(name).name == name and name.startswith("MVP-304-check-second-red-"), "Bad RED file"
        )
        value = read_file(ROOT / "docs/progress/evidence" / name)
        require(sha(value) == item["sha256"], "Retained RED artifact changed: " + name)
        archived[name] = value
    manifest_bytes = archived["MVP-304-check-second-red-manifest.json"]
    manifest = validator.parse_json(manifest_bytes)
    require(
        manifest["run_id"] == CHECK_RUN_ID and manifest["successful"] is False, "Wrong RED manifest"
    )
    require(
        [item["command"] for item in manifest["commands"]]
        == [list(command) for command in validator.EXPECTED_COMMANDS[:9]],
        "RED manifest is not the original nine actual commands",
    )
    require(
        [item["exit_code"] for item in manifest["commands"]] == [0] * 8 + [1], "RED exits changed"
    )
    require(
        read_file(ROOT / ".runtime/quality" / CHECK_RUN_ID / "manifest.json") == manifest_bytes,
        "Original run manifest no longer equals its retained RED copy",
    )
    environment_bytes = archived["MVP-304-check-second-red-environment.json"]
    validator.validate_environment(validator.parse_json(environment_bytes), manifest)
    require(
        read_file(ROOT / ".runtime/quality" / CHECK_RUN_ID / "environment.json")
        == environment_bytes,
        "Original launch environment changed",
    )
    for entry in manifest["commands"]:
        require(
            read_file(ROOT / ".runtime/quality" / CHECK_RUN_ID / entry["log"])
            == archived["MVP-304-check-second-red-" + entry["log"]],
            "Original command log changed",
        )
    old_source = manifest["source"]["source_sha256"]
    require(
        len(source) == len(old_source) == 201 and source.keys() == old_source.keys(),
        "Source201 set differs",
    )
    changed = [name for name in source if source[name] != old_source[name]]
    require(changed == [RESET_PATH], "Only the authorized original reset test may change")
    original_reset = archived["MVP-304-check-second-red-test_decision_trace_reset.py.txt"].decode(
        "utf-8-sig"
    )
    current_reset = read_file(ROOT / RESET_PATH).decode("utf-8-sig")
    original_tree, current_tree = ast.parse(original_reset), ast.parse(current_reset)
    unchanged_functions = {}
    for name in (
        "completed_transfer",
        "other_tenant_links",
        "test_failed_reset_restores_original_links_and_economic_history",
    ):
        original_function = next(
            node
            for node in original_tree.body
            if isinstance(node, ast.FunctionDef) and node.name == name
        )
        current_function = next(
            node
            for node in current_tree.body
            if isinstance(node, ast.FunctionDef) and node.name == name
        )
        require(
            ast.dump(original_function) == ast.dump(current_function),
            "Original function changed: " + name,
        )
        unchanged_functions[name] = True
    require(
        ast.get_source_segment(original_reset, original_function)
        == ast.get_source_segment(current_reset, current_function),
        "Original rollback node text changed",
    )
    head = subprocess.run(
        ["git", "rev-parse", "HEAD"], cwd=ROOT, capture_output=True, text=True, check=True
    ).stdout.strip()
    require(head == manifest["source"]["git_revision"], "HEAD changed before formal observation")
    backend = validator.clean(
        archived["MVP-304-check-second-red-09-uv.log"].decode("utf-8-sig")
    ).replace("\r\n", "\n")
    require(
        re.findall(r"^collected (\d+) items$", backend, re.MULTILINE) == ["1300"],
        "Collection differs",
    )
    require("1 failed, 464 passed, 1 warning in 9263.84s" in backend, "Actual RED summary missing")
    streams: dict[str, str] = {}
    current: str | None = None
    for line in backend.split("=== FAILURES", 1)[0].splitlines():
        match = re.fullmatch(
            r"apps[\\/]api[\\/]app[\\/]tests[\\/]([^\s]+\.py)\s+([.FEsxX]+).*", line
        )
        if match:
            current = match[1]
            streams[current] = match[2]
            continue
        continuation = re.fullmatch(r"\s*([.FEsxX]+)\s+\[\s*\d+%\]\s*", line)
        if continuation and current:
            streams[current] += continuation[1]
    for name, count in COMPLETED_MODULES.items():
        require(streams.get(name) == "." * count, f"Completed real module differs: {name}")
    proof_bytes = read_file(reset_path)
    reset = validator.parse_json(proof_bytes)
    expected_nodes = [
        RESET_PATH + "::test_reset_after_actual_three_phase_execution_is_repeatable_"
        "and_preserves_other_tenant",
        RESET_PATH + "::test_failed_reset_restores_original_links_and_economic_history",
    ]
    require(reset.get("exit_code") == 0, "Actual targeted reset did not exit zero")
    require(sorted(reset.get("nodes", [])) == sorted(expected_nodes), "Wrong reset nodes")
    command = reset.get("command")
    require(type(command) is list and bool(command), "Missing actual target command")
    uv_runner = command[:4] == ["uv", "run", "--frozen", "pytest"]
    local_runner = Path(command[0]).resolve() == (
        ROOT / ".venv/Scripts/python.exe"
    ).resolve() and command[1:3] == ["-m", "pytest"]
    require(uv_runner or local_runner, "Target must use frozen UV or the actual local venv pytest")
    require(
        "--no-cov" in command and "--cov" not in command,
        "Expected only the authorized no-cov target",
    )
    require(RESET_PATH in command or set(expected_nodes) <= set(command), "Reset selection missing")
    started = validator.instant(reset.get("started_at"), "target start")
    finished = validator.instant(reset.get("finished_at"), "target finish")
    require(
        validator.instant(manifest["finished_at"], "failed full finish")
        <= started
        <= finished
        <= datetime.now(UTC),
        "Targeted repair timing is inconsistent",
    )
    require(
        reset.get("source_sha256", {}).get(RESET_PATH) == source[RESET_PATH],
        "Targeted test source is stale",
    )
    log_bytes = read_file(bound_file(reset["log_file"]))
    require(sha(log_bytes) == reset["log_sha256"], "Actual reset log changed")
    reset_log = validator.clean(log_bytes.decode("utf-8-sig")).replace("\r\n", "\n")
    require(
        re.findall(r"collected (\d+) items", reset_log) == ["2"],
        "Actual reset did not collect exactly two",
    )
    for node in expected_nodes:
        require(
            node + " PASSED" in reset_log.replace("\\", "/"),
            "Actual selected reset node did not pass",
        )
    passed = re.findall(
        r"(?m)^(?:=+\s*)?2 passed(?:, \d+ warning)? in ([\d.]+)s[^\r\n]*$", reset_log
    )
    require(
        len(passed) == 1 and "FAILED " not in reset_log and "ERROR " not in reset_log,
        "Reset is not actual 2 passed",
    )
    require(
        float(passed[0]) <= (finished - started).total_seconds() + 5,
        "Target duration exceeds recorded wall",
    )
    static = reset.get("static")
    require(type(static) is list and len(static) == 3, "Need actual reset Ruff/format/mypy proofs")
    seen = set()
    for entry in static:
        require(
            entry.get("exit_code") == 0 and RESET_PATH in entry["command"],
            "Static target check failed",
        )
        value = read_file(bound_file(entry["log_file"]))
        require(sha(value) == entry["log_sha256"], "Static proof changed")
        args = entry["command"]
        if "mypy" in args:
            label, phrase = "mypy", "Success: no issues found"
        elif "format" in args and "--check" in args:
            label, phrase = "format", "1 file already formatted"
        else:
            require(
                ("ruff" in args or Path(args[0]).name.lower() == "ruff.exe") and "check" in args,
                "Unexpected static command",
            )
            label, phrase = "ruff", "All checks passed!"
        require(phrase in value.decode("utf-8-sig"), "Static success marker missing")
        seen.add(label)
    require(seen == {"ruff", "format", "mypy"}, "Static evidence coverage differs")
    return {
        "acceptance_policy": "USER_AUTHORIZED_TARGETED_MODULE_AND_DIRECT_INTEGRATION",
        "failed_full_check_retained": True,
        "full_check_verified": False,
        "failed_full_run_id": CHECK_RUN_ID,
        "failed_full_actual_result": red["actual_result"],
        "failed_full_verification_sha256": sha(red_bytes),
        "failed_full_manifest_sha256": sha(manifest_bytes),
        "source_bound_completed_modules": COMPLETED_MODULES,
        "targeted_reset_metadata_path": reset_path.relative_to(ROOT).as_posix(),
        "targeted_reset_metadata_sha256": sha(proof_bytes),
        "targeted_reset": reset,
        "head": head,
        "only_changed_source": changed,
        "unchanged_original_reset_functions": unchanged_functions,
        "original_rollback_text_unchanged": True,
        "original_full_check_formal_helper_sha256": ORIGINAL_HELPER_HASH,
    }


def verify(run_id: str, reset_path: Path, preflight: bool = False) -> dict[str, Any]:
    require(run_id == CHECK_RUN_ID, "Only the explicitly specified retained RED run is allowed")
    pinned = checked_pins()
    tool_bytes = read_file(Path(__file__).resolve())
    validator = load_validator()
    source_before = validator.current_source()
    check_proof = targeted_gate(validator, reset_path, source_before)
    baselines = load_baselines(pinned, validator)
    if preflight:
        return {
            "classification": "STATIC_TARGETED_ARTIFACT_PREFLIGHT_ONLY",
            "artifact_preflight_passed": True,
            "database_connected": False,
            "formal_final_proof_executed": False,
            "tool_sha256": sha(tool_bytes),

        )

    # No database import, engine construction or connection occurs above this line.
    sys.path.insert(0, str(ROOT / "apps/api"))
    from app.db.settings import DatabaseSettings

    check_full_snapshot(actual, validator)

    require(checked_pins() == pinned, "Baseline/tool bytes changed during the formal read")

def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--failed-check-run-id", required=True)
    parser.add_argument("--reset-evidence", required=True)
    parser.add_argument("--preflight", action="store_true")
    args = parser.parse_args()
    try:
        print(
            json.dumps(
                verify(args.failed_check_run_id, bound_file(args.reset_evidence), args.preflight),
                ensure_ascii=False,
                indent=2,
            )
        )
        return 0
    except Exception as error:
        # Driver errors may contain connection details. Record only the class;
        # explicit local proof failures are safe to report without a DSN.
        safe_message = str(error) if isinstance(error, VerificationError) else type(error).__name__
        print(
            json.dumps({"successful": False, "error": safe_message}, ensure_ascii=False),
            file=sys.stderr,
        )
        return 1


if __name__ == "__main__":
    sys.stdout.reconfigure(encoding="utf-8", errors="strict")
    sys.stderr.reconfigure(encoding="utf-8", errors="strict")
    raise SystemExit(main())
